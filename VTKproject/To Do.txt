// Inside MainWindow class in mainwindow.h

#include "VRRenderThread.h" // Include the VRRenderThread header file

private slots:
    void on_startVRButton_clicked(); // Define a slot for VR button click

private:
    VRRenderThread* vrThread; // Member variable to hold VRRenderThread instance

////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////
// Inside MainWindow class implementation in mainwindow.cpp

#include "ModelPart.h" // Include the ModelPart header file

void MainWindow::on_startVRButton_clicked() {
    vrThread = new VRRenderThread(this); // Create a new instance of VRRenderThread
    // Add actors to VRRenderThread using addActorOffline function
    for each (ModelPart* part in modelPartsList) {
        vtkActor* newActor = part->createActor(); // Assuming createActor() creates an actor for ModelPart
        vrThread->addActorOffline(newActor);
    }
    vrThread->start(); // Start the VR thread
}

planeLeft->SetNormal(-1.0, 0.0, 0.0);

vtkSmartPointer<vtkClipDataSet> clipFilter = vtkSmartPointer<vtkClipDataSet>::New();
clipFilter->SetInputConnection(SOURCE->GetOutputPort());
clipFilter->SetClipFunction(planeLeft);

MAPPER->SetInputConnection(clipFilter->GetOutputPort());

//Shrink Filter:

shrinkFilter->Update();

MAPPER->SetInputConnection(shrinkFilter->GetOutputPort());
